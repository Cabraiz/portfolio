$paths = @(
  "${env:ProgramFiles}\Fortinet\FortiClient\FortiClient.exe",
  "${env:ProgramFiles}\Fortinet\FortiClient\FortiVPN.exe",
  "${env:ProgramFiles(x86)}\Fortinet\FortiClient\FortiClient.exe",
  "${env:ProgramFiles(x86)}\Fortinet\FortiClient\FortiVPN.exe"
)

foreach ($p in $paths) {
  if (Test-Path $p) {
    Write-Host "`nEncontrado:" $p
    (Get-Item $p).VersionInfo | Select-Object `
      ProductName,
      ProductVersion,
      FileVersion,
      OriginalFilename
  }
}
